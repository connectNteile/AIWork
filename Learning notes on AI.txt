- create a module for string operation including function to reverse string, count words /vowel, check for Palindromes . import into script and test the function.

--- solution ---
def check_vowels(text):
    # Define a set of vowels for fast lookup
    vowels = "aeiouAEIOU"
    
    # Extract all vowels found in the string
    found_vowels = [char for char in text if char in vowels]
    
    return {
        "has_vowels": len(found_vowels) > 0,
        "count": len(found_vowels),
        "list": found_vowels
    }

# Example usage:
text_to_check = "Congratulations"
result = check_vowels(text_to_check)

print(f"Contains vowels?: {result['has_vowels']}")
print(f"Total vowel count: {result['count']}")
print(f"Vowels found: {result['list']}")
